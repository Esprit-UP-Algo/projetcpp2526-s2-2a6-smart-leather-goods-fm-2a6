Le code a été simplifié pour garantir la compatibilité:

```cpp
auto majPhotoCuir = [=](const QString &matiere) {
    // 1. Définir le chemin de base
    QString chemin = ":/pic vachette.jpg";  // Par défaut
    
    if (matiere == "Cuir Agneau") {
        chemin = ":/pic agneau.jpg";
    } else if (matiere == "Cuir Veau") {
        chemin = ":/pic veau.jpg";  // ✓ Corrigé
    }
    
    // 2. Essayer ressource Qt
    QPixmap pix(chemin);
    
    // 3. Si échec, essayer fichiers disque
    if (pix.isNull()) {
        QStringList paths;
        paths << baseName;
        paths << ("../" + baseName);
        paths << ("images/" + baseName);
        
        // Cas spécial pour veau
        if (matiere == "Cuir Veau") {
            paths << "cuir veau.jpg";
            paths << "../cuir veau.jpg";
        }
        
        for (const QString &path : paths) {
            pix = QPixmap(path);
            if (!pix.isNull()) break;
        }
    }
    
    // 4. Afficher ou message d'erreur
    if (!pix.isNull()) {
        photoCuir->setPixmap(pix.scaled(...));
    } else {
        photoCuir->setText("Image indisponible");
    }
};
```
